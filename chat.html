Certainly. Here is a Python script using the pandas library to parse your CSV and calculate the average age.

python 
import pandas as pd

def calculate_average_age(csv_file_path):
    try:
        # Load the CSV data into a pandas DataFrame
        df = pd.read_csv(csv_file_path)
        
        # Check if 'age' column exists
        if 'age' not in df.columns.str.lower():
            return "Error: 'age' column not found in the CSV."
            
        # Ensure 'age' is numeric, coercing errors to NaN, then drop NaNs
        df['age'] = pd.to_numeric(df['age'], errors='coerce')
        valid_ages = df['age'].dropna()
        
        if valid_ages.empty:
            return "Error: No valid age data found."
            
        # Calculate and return the mean
        avg_age = valid_ages.mean()
        return round(avg_age, 2)
        
    except FileNotFoundError:
        return "Error: File not found."
    except Exception as e:
        return f"An unexpected error occurred: {e}"
